calcFlow(float src, int len) =>
    roc = ta.roc(src, len)
    ta.ema(roc, len)

// Adaptive: flow as a multiple of the symbol's own typical move, so one threshold fits every
// timeframe and market. Fixed: raw % flow (v4.5 behaviour).
calcFlowOffset(int len, int off) =>
    f = calcFlow(close, len)
    scale = ta.sma(math.abs(ta.roc(close, len)), 100)
    v = adaptiveThreshold ? (scale > 0 ? f / scale : 0.0) : f
    [v[off], time[off]]

// Non-repainting request: higher timeframes use the last CLOSED bar ([1] + lookahead_on),
// so history and live bars show the same values. Lower timeframes are already closed bars.
// Optionally, closed markets (weekends, outside US hours) count as neutral instead of their last reading.
getFlow(simple string sym, simple string tf) =>
    bool htf = timeframe.in_seconds(tf) >= timeframe.in_seconds()
    int off = htf ? 1 : 0
    // NOTE: calcFlowOffset reads adaptiveThreshold, which depends on the timeframe. Inside
    // request.security Pine re-evaluates it in the REQUESTED context, so lower-timeframe reads
    // (e.g. 15m on a 3H/4H chart) can come back volatility-scaled. The v4.6.0 backtests depend on
    // this; replacing it with an explicit per-timeframe rule changed 3H from +47% to +19% (see
    // lab/README.md). Do not "simplify" this without re-running the strategy.
    [f, t] = request.security(sym, tf, calcFlowOffset(flowLength, off), lookahead=htf ? barmerge.lookahead_on : barmerge.lookahead_off, ignore_invalid_symbol=true)
    stale = neutralWhenClosed and (na(t) or time - t > (3 * timeframe.in_seconds(tf) + timeframe.in_seconds()) * 1000)
    stale ? 0.0 : nz(f)

getSignal(float flow, bool invert) =>
    threshold = adaptiveThreshold ? adaptiveLevel : 0.5
    rawSignal = flow > threshold ? 1.0 : flow < -threshold ? -1.0 : 0.0
    invert ? -rawSignal : rawSignal

calcCorrelation(float signal, int len) =>
    // Threshold for combined multi-TF signals (0.3 = need 30%+ agreement)
    // This filters out weak/conflicting signals that produce noisy correlation
    signalThreshold = 0.3

    priceUp = close > close[1]
    signalBull = signal > signalThreshold
    signalBear = signal < -signalThreshold
    signalActive = math.abs(signal) > signalThreshold

    correctBull = signalBull and priceUp ? 1.0 : 0.0
    correctBear = signalBear and not priceUp ? 1.0 : 0.0
    correct = correctBull + correctBear
    active = signalActive ? 1.0 : 0.0

    correctSum = math.sum(correct, len)
    activeSum = math.sum(active, len)

    activeSum > 10 ? (correctSum / activeSum) * 100 : 50.0

calcEdge(float corr) =>
    math.abs(corr - 50) / 50

calcMultiplier(float corr) =>
    corr >= 50 ? 1.0 : -1.0

calcWeight(float corr) =>
    0.2 + 0.8 * calcEdge(corr)

// Helper: Convert bars to human-readable time
getBarsAsTime(int bars) =>
    tfSeconds = timeframe.in_seconds()
    totalMinutes = bars * tfSeconds / 60
    if totalMinutes < 60
        str.tostring(math.round(totalMinutes)) + "m"
    else if totalMinutes < 1440
        str.tostring(math.round(totalMinutes / 60, 1)) + "h"
    else
        str.tostring(math.round(totalMinutes / 1440, 1)) + "d"

// Helper: Get signal arrow with color context (no more ↻)
getArrow(float sig, float mult) =>
    eff = sig * mult
    eff > 0.3 ? "▲" : eff < -0.3 ? "▼" : "•"

// Helper: Build confidence bar (simpler)
getConfBar(float score) =>
    filled = math.round(math.abs(score) * 10)
    bar = ""
    for i = 1 to 10
        bar := bar + (i <= filled ? "█" : "░")
    bar

// Helper: Get accuracy icon based on correlation %
getAccuracyIcon(float corr) =>
    corr > 60 ? "✓" : corr < 40 ? "✗" : "~"

// Helper: Get streak text (e.g., "3W" or "2L")
getStreakText(int streak) =>
    math.abs(streak) < 2 ? "" : streak > 0 ? str.tostring(streak) + "W" : str.tostring(math.abs(streak)) + "L"

// STRATEGY - BACKTEST

btDirection = input.string("Long Only", "Trade Direction", options=["Both", "Long Only", "Short Only"], group="Backtest")
btEntry = input.string("Standard (50%+)", "Entry Signal", options=["Flip", "Standard (50%+)", "Full Send (70%+)"], group="Backtest",
     tooltip="Flip: enter on each confirmed capital flow flip\nStandard / Full Send: enter only when the score also reaches that strength")
btExit = input.string("Opposite Flip", "Exit", options=["Opposite Flip", "Score Back To Neutral"], group="Backtest",
     tooltip="Opposite Flip: stay in until flow flips the other way (stop-and-reverse when Both)\nScore Back To Neutral: exit once the score falls back inside +/- the flip threshold")
btStopAtr = input.float(0.0, "Stop Loss (x ATR, 0 = off)", minval=0.0, maxval=10.0, step=0.5, group="Backtest")
btTargetAtr = input.float(0.0, "Take Profit (x ATR, 0 = off)", minval=0.0, maxval=20.0, step=0.5, group="Backtest")
btStart = input.time(timestamp("2024-01-01T00:00:00"), "Backtest From", group="Backtest")

inWindow = time >= btStart
allowLong = btDirection != "Short Only"
allowShort = btDirection != "Long Only"
entryLevel = btEntry == "Full Send (70%+)" ? 0.7 : btEntry == "Standard (50%+)" ? 0.5 : 0.0

// Flip entries use the indicator's confirmed flips; strength entries need the flip direction AND the score level
var int btDir = 0
if flipToBullish
    btDir := 1
if flipToBearish
    btDir := -1
longSignal = btEntry == "Flip" ? flipToBullish : btDir == 1 and normalizedScore > entryLevel and normalizedScore[1] <= entryLevel
shortSignal = btEntry == "Flip" ? flipToBearish : btDir == -1 and normalizedScore < -entryLevel and normalizedScore[1] >= -entryLevel

btAtr = ta.atr(14)

if inWindow and longSignal and allowLong
    strategy.entry("Long", strategy.long)
if inWindow and shortSignal and allowShort
    strategy.entry("Short", strategy.short)

// Exits
exitLong = btExit == "Opposite Flip" ? flipToBearish : normalizedScore < flipLevel
exitShort = btExit == "Opposite Flip" ? flipToBullish : normalizedScore > -flipLevel
if strategy.position_size > 0 and exitLong and not (shortSignal and allowShort)
    strategy.close("Long", comment="Exit")
if strategy.position_size < 0 and exitShort and not (longSignal and allowLong)
    strategy.close("Short", comment="Exit")

// Optional ATR stop / target, set from the entry price
if strategy.position_size > 0 and (btStopAtr > 0 or btTargetAtr > 0)
    strategy.exit("Long SL/TP", "Long",
         stop=btStopAtr > 0 ? strategy.position_avg_price - btStopAtr * btAtr : na,
         limit=btTargetAtr > 0 ? strategy.position_avg_price + btTargetAtr * btAtr : na)
if strategy.position_size < 0 and (btStopAtr > 0 or btTargetAtr > 0)
    strategy.exit("Short SL/TP", "Short",
         stop=btStopAtr > 0 ? strategy.position_avg_price + btStopAtr * btAtr : na,
         limit=btTargetAtr > 0 ? strategy.position_avg_price - btTargetAtr * btAtr : na)
